cmake_minimum_required(VERSION 3.22)
project(gz-fuel_tools VERSION 12.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
include(GNUInstallDirs)
find_package(gz-common REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
find_program(CARGO cargo REQUIRED)
set(UPSTREAM_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../gz-fuel-tools" CACHE PATH "Unmodified upstream source for parity tests")
set(PROJECT_VERSION_FULL "12.0.0~pre2")
set(PROJECT_BINARY_PATH "${CMAKE_CURRENT_BINARY_DIR}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/gz/fuel_tools")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/cmake/config.hh.in" "include/gz/fuel_tools/config.hh")
configure_file(cmake/Export.hh.in "include/gz/fuel_tools/Export.hh" COPYONLY)
file(GLOB PUBLIC_HEADERS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/include" "include/gz/fuel_tools/*.hh")
set(UMBRELLA "// Copyright (C) 2026 Open Source Robotics Foundation\n// SPDX-License-Identifier: Apache-2.0\n#pragma once\n#include <gz/fuel_tools/config.hh>\n")
foreach(HEADER ${PUBLIC_HEADERS})
  string(APPEND UMBRELLA "#include <${HEADER}>\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/include/gz/fuel_tools.hh" CONTENT "${UMBRELLA}" @ONLY)
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS "src/*.rs" "src/metadata/*")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/cargo/release/libgz_fuel_tools.a" "${CMAKE_CURRENT_BINARY_DIR}/cargo/release/gz-fuel"
 COMMAND ${CMAKE_COMMAND} -E env CARGO_TARGET_DIR=${CMAKE_CURRENT_BINARY_DIR}/cargo ${CARGO} build --release --locked --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml"
  DEPENDS ${RUST_SOURCES} Cargo.toml Cargo.lock VERBATIM)
add_custom_target(fuel-rust DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/cargo/release/libgz_fuel_tools.a" "${CMAKE_CURRENT_BINARY_DIR}/cargo/release/gz-fuel")
add_library(gz-fuel_tools SHARED cpp/Bridge.cc cpp/Core.cc cpp/Json.cc cpp/Io.cc cpp/Cache.cc cpp/Client.cc cpp/Pager.cc cpp/Iterators.cc cpp/Interface.cc cpp/Command.cc)
add_dependencies(gz-fuel_tools fuel-rust)
foreach(alias IN ITEMS gz-fuel_tools::gz-fuel_tools gz-fuel_tools::core gz-fuel_tools::requested gz-fuel_tools::all)
  add_library(${alias} ALIAS gz-fuel_tools)
endforeach()
set_target_properties(gz-fuel_tools PROPERTIES OUTPUT_NAME gz-fuel_tools SOVERSION 12 VERSION 12.0.0 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/src")
target_include_directories(gz-fuel_tools PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE cpp)
target_link_libraries(gz-fuel_tools PUBLIC gz-common::gz-common PRIVATE PkgConfig::JSONCPP "${CMAKE_CURRENT_BINARY_DIR}/cargo/release/libgz_fuel_tools.a" ssl crypto dl pthread)
option(FUEL_CPP_COVERAGE "Instrument C++ facade coverage" OFF)
if(FUEL_CPP_COVERAGE)
 target_compile_options(gz-fuel_tools PRIVATE --coverage -O0 -g)
 target_link_options(gz-fuel_tools PRIVATE --coverage)
endif()
if(BUILD_TESTING)
  option(FUEL_RUN_DISABLED_TESTS "Run inherited disabled upstream service tests" ON)
  find_package(gz-common REQUIRED COMPONENTS testing)
  find_package(gz-msgs REQUIRED)
  set(PROJECT_SOURCE_DIR "${UPSTREAM_SOURCE}")
  configure_file("${UPSTREAM_SOURCE}/test/test_config.hh.in" "include/test_config.hh")
  add_subdirectory("${UPSTREAM_SOURCE}/test/gtest_vendor" "gtest_vendor" EXCLUDE_FROM_ALL)
  file(GLOB PARITY_TESTS "${UPSTREAM_SOURCE}/src/*_TEST.cc")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/bin/gz" CONTENT "#!/bin/sh\nexec \"${CMAKE_CURRENT_BINARY_DIR}/cargo/release/gz-fuel\" \"$@\"\n" @ONLY)
  file(CHMOD "${CMAKE_CURRENT_BINARY_DIR}/bin/gz" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  set(GZ_PROGRAM "${CMAKE_CURRENT_BINARY_DIR}/bin/gz")
  foreach(TEST_SOURCE ${PARITY_TESTS})
    get_filename_component(TEST_NAME ${TEST_SOURCE} NAME_WE)
    add_executable(UNIT_${TEST_NAME} ${TEST_SOURCE})
    target_include_directories(UNIT_${TEST_NAME} PRIVATE cpp "${CMAKE_CURRENT_BINARY_DIR}/include")
    target_link_libraries(UNIT_${TEST_NAME} PRIVATE gz-fuel_tools gtest gtest_main gz-common::testing gz-msgs::gz-msgs)
    target_compile_definitions(UNIT_${TEST_NAME} PRIVATE GZ_PATH="${GZ_PROGRAM}" TESTING_PROJECT_SOURCE_DIR="${UPSTREAM_SOURCE}")
    add_test(NAME UNIT_${TEST_NAME} COMMAND UNIT_${TEST_NAME})
    set_tests_properties(UNIT_${TEST_NAME} PROPERTIES TIMEOUT 1000 RUN_SERIAL TRUE ENVIRONMENT "LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}/src:$ENV{LD_LIBRARY_PATH}")
    if(FUEL_RUN_DISABLED_TESTS AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
      set_property(TEST UNIT_${TEST_NAME} APPEND PROPERTY ENVIRONMENT "GTEST_ALSO_RUN_DISABLED_TESTS=1")
    endif()
  endforeach()
  file(GLOB SUPPLEMENTARY_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/*compatibility.cc")
  foreach(TEST_SOURCE ${SUPPLEMENTARY_TESTS})
    get_filename_component(TEST_NAME ${TEST_SOURCE} NAME_WE)
    add_executable(SUPPLEMENTARY_${TEST_NAME} ${TEST_SOURCE})
    target_include_directories(SUPPLEMENTARY_${TEST_NAME} PRIVATE cpp "${CMAKE_CURRENT_BINARY_DIR}/include")
    target_link_libraries(SUPPLEMENTARY_${TEST_NAME} PRIVATE gz-fuel_tools gtest gtest_main)
    add_test(NAME SUPPLEMENTARY_${TEST_NAME} COMMAND SUPPLEMENTARY_${TEST_NAME})
    set_tests_properties(SUPPLEMENTARY_${TEST_NAME} PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE ENVIRONMENT "LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}/src:$ENV{LD_LIBRARY_PATH}")
  endforeach()
  add_executable(INTEGRATION_zip "${UPSTREAM_SOURCE}/test/integration/zip.cc")
  target_include_directories(INTEGRATION_zip PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/include")
  target_link_libraries(INTEGRATION_zip PRIVATE gz-fuel_tools gtest gtest_main)
  add_test(NAME INTEGRATION_zip COMMAND INTEGRATION_zip)
  set_tests_properties(INTEGRATION_zip PROPERTIES RUN_SERIAL TRUE ENVIRONMENT "LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}/src:$ENV{LD_LIBRARY_PATH}")
endif()

install(FILES "${CMAKE_CURRENT_BINARY_DIR}/gz-fuel_tools-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/gz-fuel_tools-config-version.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/gz-fuel_tools")
